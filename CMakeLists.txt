cmake_minimum_required(VERSION 3.24)
project(SimpleAVShow VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(SAV_BUILD_APP "Build the native desktop editor" ON)
set(SAV_FFMPEG_ROOT "" CACHE PATH "Use an existing FFmpeg SDK (include/, lib/, bin/) instead of system libraries")
option(SAV_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
option(SAV_TSAN "Enable the thread sanitizer" OFF)
if(SAV_SANITIZE AND SAV_TSAN)
  message(FATAL_ERROR "SAV_SANITIZE and SAV_TSAN cannot be combined; ThreadSanitizer is incompatible with AddressSanitizer.")
endif()
if(SAV_TSAN AND MSVC)
  message(FATAL_ERROR "SAV_TSAN requires Clang or GCC.")
endif()
if(SAV_SANITIZE AND NOT MSVC)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
if(SAV_TSAN)
  add_compile_options(-fsanitize=thread -fno-omit-frame-pointer -g)
  add_link_options(-fsanitize=thread)
endif()
find_package(Threads REQUIRED)
add_library(sav_core src/core/Transport.cpp src/core/AudioBuffer.cpp)
target_link_libraries(sav_core PUBLIC Threads::Threads)
target_include_directories(sav_core PUBLIC src)
if(MSVC)
  target_compile_options(sav_core PRIVATE /W4)
else()
  target_compile_options(sav_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_executable(transport_tests tests/transport_tests.cpp)
target_link_libraries(transport_tests PRIVATE sav_core)
add_test(NAME transport COMMAND transport_tests)

if(SAV_BUILD_APP)
  set(CMAKE_AUTOMOC ON)
  find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets Concurrent)
  # Homebrew's libarchive is keg-only; prefer it to the older macOS SDK library.
  if(APPLE AND EXISTS "/opt/homebrew/opt/libarchive")
    list(APPEND CMAKE_PREFIX_PATH "/opt/homebrew/opt/libarchive")
  endif()
  find_package(LibArchive 3.6 REQUIRED)
  find_package(PkgConfig QUIET)
  if(PkgConfig_FOUND)
    pkg_check_modules(FFMPEG QUIET IMPORTED_TARGET libavformat libavcodec libavutil libswresample libswscale)
    pkg_check_modules(RTAUDIO QUIET IMPORTED_TARGET rtaudio>=6.0)
    pkg_check_modules(RTMIDI QUIET IMPORTED_TARGET rtmidi)
  endif()
  if(SAV_FFMPEG_ROOT)
    find_path(FFMPEG_INCLUDE_DIR libavformat/avformat.h PATHS "${SAV_FFMPEG_ROOT}/include" NO_DEFAULT_PATH REQUIRED)
    foreach(lib avformat avcodec avutil swresample swscale)
      find_library(FFMPEG_${lib}_LIBRARY NAMES ${lib} PATHS "${SAV_FFMPEG_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
      list(APPEND SAV_FFMPEG ${FFMPEG_${lib}_LIBRARY})
    endforeach()
  elseif(TARGET PkgConfig::FFMPEG)
    set(SAV_FFMPEG PkgConfig::FFMPEG)
  else()
    find_path(FFMPEG_INCLUDE_DIR libavformat/avformat.h REQUIRED)
    foreach(lib avformat avcodec avutil swresample swscale)
      find_library(FFMPEG_${lib}_LIBRARY NAMES ${lib} REQUIRED)
      list(APPEND SAV_FFMPEG ${FFMPEG_${lib}_LIBRARY})
    endforeach()
  endif()
  if(TARGET PkgConfig::RTAUDIO)
    set(SAV_RTAUDIO PkgConfig::RTAUDIO)
  else()
    find_package(RtAudio CONFIG REQUIRED)
    set(SAV_RTAUDIO RtAudio::rtaudio)
  endif()
  if(TARGET PkgConfig::RTMIDI)
    set(SAV_RTMIDI PkgConfig::RTMIDI)
  else()
    find_package(RtMidi CONFIG REQUIRED)
    set(SAV_RTMIDI RtMidi::rtmidi)
    get_target_property(RTMIDI_INCLUDE_HINTS RtMidi::rtmidi INTERFACE_INCLUDE_DIRECTORIES)
    find_path(RTMIDI_HEADER_DIR RtMidi.h HINTS ${RTMIDI_INCLUDE_HINTS} PATH_SUFFIXES rtmidi REQUIRED)
  endif()
  add_library(sav_desktop
    src/app/Project.cpp src/app/ProjectPackage.cpp src/app/Controller.cpp
    src/app/Timeline.cpp src/app/ProgramView.cpp src/app/MainWindow.cpp
    src/app/MidiMessageEditor.cpp
    src/media/Media.cpp src/media/AudioSources.cpp src/media/AudioDevice.cpp src/media/MidiOutput.cpp
  )
  target_include_directories(sav_desktop PUBLIC src ${RTMIDI_HEADER_DIR} PRIVATE ${FFMPEG_INCLUDE_DIR})
  target_link_libraries(sav_desktop PUBLIC sav_core Qt6::Widgets Qt6::Concurrent ${SAV_RTAUDIO} ${SAV_RTMIDI} PRIVATE ${SAV_FFMPEG} LibArchive::LibArchive)
  if(MSVC)
    target_compile_options(sav_desktop PUBLIC /utf-8)
    target_compile_definitions(sav_desktop PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
  else()
    target_compile_options(sav_desktop PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  add_executable(simpleavshow WIN32 MACOSX_BUNDLE src/app/main.cpp)
  target_link_libraries(simpleavshow PRIVATE sav_desktop)
  # --version reports the project version, so packages and the app cannot drift apart.
  target_compile_definitions(simpleavshow PRIVATE SAV_VERSION="${PROJECT_VERSION}")
  if(WIN32)
    target_link_libraries(simpleavshow PRIVATE winmm)
  endif()
  find_package(Qt6 REQUIRED COMPONENTS Test)
  add_executable(app_tests tests/app_tests.cpp)
  target_link_libraries(app_tests PRIVATE sav_desktop Qt6::Test LibArchive::LibArchive)
  add_test(NAME app COMMAND app_tests)
  set_tests_properties(app PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  set_target_properties(simpleavshow PROPERTIES MACOSX_BUNDLE_GUI_IDENTIFIER "org.simpleavshow.app" MACOSX_BUNDLE_BUNDLE_NAME "SimpleAVShow")
  install(TARGETS simpleavshow BUNDLE DESTINATION . RUNTIME DESTINATION bin)
  if(WIN32)
    if(SAV_FFMPEG_ROOT)
      install(DIRECTORY "${SAV_FFMPEG_ROOT}/bin/" DESTINATION bin FILES_MATCHING PATTERN "*.dll")
    endif()
    if(DEFINED VCPKG_INSTALLED_DIR)
      install(DIRECTORY "${VCPKG_INSTALLED_DIR}/${VCPKG_TARGET_TRIPLET}/bin/" DESTINATION bin FILES_MATCHING PATTERN "*.dll")
    endif()
    qt_generate_deploy_app_script(TARGET simpleavshow OUTPUT_SCRIPT qt_deploy NO_UNSUPPORTED_PLATFORM_ERROR)
    install(SCRIPT ${qt_deploy})
  endif()
endif()
